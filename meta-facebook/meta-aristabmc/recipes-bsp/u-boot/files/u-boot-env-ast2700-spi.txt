baudrate=9600
bootargs=console=ttyS12,9600n8 root=/dev/ram rw earlycon
bootcmd=run bootspi
bootdelay=2
ftdspiaddr=0x100420000
bootspi=fdt addr ${ftdspiaddr} && fdt header get fitsize totalsize && cp.b ${ftdspiaddr} ${loadaddr} ${fitsize} && bootm; echo Error loading kernel FIT image
loadaddr=0x403000000
verify=yes
